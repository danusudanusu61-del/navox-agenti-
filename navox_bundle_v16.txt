    static int lev(String a, String b) {
        int m = a.length(), n = b.length();
        int[] p = new int[n + 1];
        for (int j = 0; j <= n; j++) p[j] = j;
        for (int i = 1; i <= m; i++) {
            int[] q = new int[n + 1];
            q[0] = i;
            for (int j = 1; j <= n; j++) {
                int cost = a.charAt(i - 1) == b.charAt(j - 1) ? 0 : 1;
                q[j] = Math.min(Math.min(p[j] + 1, q[j - 1] + 1), p[j - 1] + cost);
            }
            p = q;
        }
        return p[n];
    }

        Matcher m;

        }
    };

        this.cb = cb;
    }

                new StorageService.Callback<Model>() {
                    public void onComplete(Model m) {
                        if (dead) return;
                        model = m;
                        try {
                            rec = new Recognizer(model, 16000.0f, GRAMMAR);
                            resume();
                        } catch (Exception e) {
                            cb.onFail();
                        }
                    }
                },

                    public void onComplete(IOException e) {
                        cb.onFail();
                    }
                });
    }

        if (ss != null) {
            try {
                ss.stop();
            } catch (Exception e) { }
            try {
                ss.shutdown();
            } catch (Exception e) { }
            ss = null;
        }
    }

    public void resume() {
        if (dead || rec == null) return;
        fired = false;
        try {
            stopService();
            rec.reset();
            ss = new SpeechService(rec, 16000.0f);
            ss.startListening(this);
            tries = 0;
            if (!ready) {
                ready = true;
                cb.onReady();
            }
        } catch (Exception e) {
            tries++;
            if (tries < 6) h.postDelayed(resumeR, 1500);
            else cb.onFail();
        }
    }

    public void pause() {
        h.removeCallbacks(resumeR);
        stopService();
    }

    public void destroy() {
        dead = true;
        pause();
        try {
            if (rec != null) rec.close();
        } catch (Exception e) { }
        try {
            if (model != null) model.close();
        } catch (Exception e) { }
    }

            }
        } catch (Exception e) { }
    }

    static void loop(Context c, String goal, String proxy, String token) throws Exception {

            if (act.equals("done")) {
                emit(c, a.optString("say", "Done."), true);
                return;
            }
            String sig = a.toString();
            rep = sig.equals(lastSig) ? rep + 1 : 0;
            lastSig = sig;
            if (rep >= 2) {
                emit(c, "I am stuck on this screen. Please continue yourself.", true);
                return;

            emit(c, step + ". " + a.optString("why", act), false);

                return;

    static String doAction(Context c, JSONObject a, Screen s) {
        String act = a.optString("action", "");
        if (act.equals("open_app")) {

            if (ap != null && CommandRunner.launch(c, ap.pkg)) return "opened " + ap.label;
            return "could not open " + nm;

        if (act.equals("open_url")) {
            CommandRunner.view(c, a.optString("url", ""), null);
            return "opened url";
        }
        if (act.equals("back")) {
            NavoxAccess.global(AccessibilityService.GLOBAL_ACTION_BACK);
            return "went back";
        }
        if (act.equals("home")) {
            NavoxAccess.global(AccessibilityService.GLOBAL_ACTION_HOME);
            return "went home";
        }
        if (act.equals("wait")) return "waited";
        if (act.equals("enter")) {
            NavoxAccess.imeEnter();
            return "pressed enter";
        }
        if (act.equals("scroll")) {
            boolean down = !"up".equals(a.optString("dir", "down"));
            for (AccessibilityNodeInfo n : s.nodes) {
                if (n.isScrollable()) {
                    n.performAction(down ? AccessibilityNodeInfo.ACTION_SCROLL_FORWARD : AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD);
                    return "scrolled " + (down ? "down" : "up");
                }
            }
            return "nothing to scroll";
        }
        int id = a.optInt("id", -1);
        if (id < 0 || id >= s.nodes.size()) return "invalid element id";
        AccessibilityNodeInfo n = s.nodes.get(id);
        String label = s.labels.get(id);
        if (act.equals("tap")) {
            if (RISKY.matcher(label.trim()).find()) {
                return "STOP:I prepared everything. Please tap \"" + label + "\" yourself.";
            }
            NavoxAccess.tap(n);
            return "tapped " + label;
        }
        if (act.equals("type")) {
            if (n.isPassword()) return "STOP:I will not type into password fields.";
            Bundle b = new Bundle();
            b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, a.optString("text", ""));
            n.performAction(AccessibilityNodeInfo.ACTION_CLICK);
            n.performAction(AccessibilityNodeInfo.ACTION_FOCUS);
            n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, b);
            return "typed text";
        }
        return "unknown action";
    }

    static String trim(String s, int n) {
        String t = s.replace('\n', ' ').trim();
        return t.length() > n ? t.substring(0, n) : t;
    }

    static Screen dump() {
        Screen s = new Screen();
        NavoxAccess svc = NavoxAccess.inst;
        AccessibilityNodeInfo root = svc == null ? null : svc.getRootInActiveWindow();
        if (root == null) return s;
        s.pkg = String.valueOf(root.getPackageName());
        List<AccessibilityNodeInfo> q = new ArrayList<AccessibilityNodeInfo>();
        q.add(root);
        int i = 0;
        while (i < q.size() && i < 800 && s.nodes.size() < 90) {
            AccessibilityNodeInfo n = q.get(i++);
            if (n == null) continue;
            for (int k = 0; k < n.getChildCount(); k++) {
                AccessibilityNodeInfo ch = n.getChild(k);
                if (ch != null) q.add(ch);
            }
            if (!n.isVisibleToUser()) continue;
            CharSequence tx = n.getText();
            CharSequence ds = n.getContentDescription();
            String label = (tx != null && tx.length() > 0) ? tx.toString() : (ds != null ? ds.toString() : "");
            boolean act = n.isClickable() || n.isEditable() || n.isScrollable() || n.isCheckable();
            if (label.length() == 0 && !act) continue;
            if (n.isPassword()) label = "(password field)";
            String kind = n.isEditable() ? "input" : n.isScrollable() ? "scroll" : n.isClickable() ? "button" : "text";
            if (kind.equals("text") && label.length() == 0) continue;
            s.text.append("[").append(s.nodes.size()).append("] ").append(kind).append(": ").append(trim(label, labelMax));
            if (n.isChecked()) s.text.append(" (on)");
            s.text.append("\n");
            s.nodes.add(n);
            s.labels.add(label);
        }
        return s;
    }

    static JSONObject parse(String s) throws Exception {
        int a = s.indexOf('{');
        int b = s.lastIndexOf('}');
        if (a < 0 || b <= a) throw new Exception("unclear reply from AI");
        return new JSONObject(s.substring(a, b + 1));
    }

    static String ask(String proxy, String token, String user) throws Exception {
        JSONObject body = new JSONObject();
        body.put("system", SYS);
        body.put("max_tokens", 400);
        JSONArray msgs = new JSONArray();
        JSONObject m = new JSONObject();
        m.put("role", "user");
        m.put("content", user);
        msgs.put(m);
        body.put("messages", msgs);
        HttpURLConnection con = (HttpURLConnection) new URL(proxy).openConnection();
        con.setRequestMethod("POST");
        con.setConnectTimeout(15000);
        con.setReadTimeout(45000);
        con.setRequestProperty("content-type", "application/json");
        con.setRequestProperty("x-app-token", token);
        con.setDoOutput(true);
        OutputStream os = con.getOutputStream();
        os.write(body.toString().getBytes("UTF-8"));
        os.close();
        int code = con.getResponseCode();
        InputStream is = code >= 400 ? con.getErrorStream() : con.getInputStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[4096];
        int n;
        while (is != null && (n = is.read(buf)) > 0) bo.write(buf, 0, n);
        JSONObject r = new JSONObject(bo.toString("UTF-8"));
        if (code >= 400) {
            JSONObject er = r.optJSONObject("error");
            throw new Exception(er != null ? er.optString("message", "HTTP " + code) : "HTTP " + code);
        }
        return r.getJSONArray("content").getJSONObject(0).optString("text", "");
    }

    public static void init(Context ctx) {

            public void onInit(int st) {

                    return;
                }
                try {
                    tts.setSpeechRate(0.93f);

    static void pickVoice(Locale loc) {
        try {
            int r = tts.setLanguage(loc);
            if (r < 0) {
                loc = new Locale("en", "IN");
                tts.setLanguage(loc);
            }
            android.speech.tts.Voice best = null;
            int bq = -1;
            Set<android.speech.tts.Voice> all = tts.getVoices();
            if (all != null) {
                for (android.speech.tts.Voice v : all) {
                    Locale l = v.getLocale();
                    if (!l.getLanguage().equals(loc.getLanguage())) continue;
                    if (v.isNetworkConnectionRequired()) continue;
                    int q = v.getQuality();
                    if (l.getCountry().equals(loc.getCountry())) q += 50;
                    if (q > bq) {
                        bq = q;
                        best = v;
                    }
                }
            }
            if (best != null) {
                tts.setVoice(best);
                voiceName = best.getName();
            }
        } catch (Throwable e) { }
    }

    public static void speak(final Context ctx, final String text, final String lang) {
        if (text == null || text.trim().length() == 0) return;
        final Context app = ctx.getApplicationContext();
        final String t = text.trim();
        final int my = ++gen;
        stopPlayer();
        if (!cloud(app) || t.length() > 180) {
            local(app, t, lang);
            return;
        }
        new Thread(new Runnable() {
            public void run() {
                final File[] out = new File[1];
                Thread ft = new Thread(new Runnable() {
                    public void run() {
                        try {
                            out[0] = fetch(app, t);
                        } catch (Throwable e) {
                            lastErr = String.valueOf(e.getMessage());
                        }
                    }
                });
                ft.start();
                try {
                    ft.join(1500);
                } catch (Exception e) { }
                File f = out[0];
                if (f != null && my == gen) {
                    try {
                        play(f);
                        status = "cloud human voice OK";
                        return;
                    } catch (Throwable e) {
                        lastErr = String.valueOf(e.getMessage());
                    }
                }

    static String md5(String s) throws Exception {

    static File fetch(Context ctx, String text) throws Exception {
        File dir = new File(ctx.getCacheDir(), "tts");
        dir.mkdirs();
        File f = new File(dir, md5(text) + ".wav");
        if (f.exists() && f.length() > 1000) return f;
        android.content.SharedPreferences sp = ctx.getSharedPreferences("navox", 0);
        String proxy = sp.getString("proxy", "");
        String token = sp.getString("ptoken", "");
        if (proxy.length() == 0) {
            lastErr = "no proxy";
            return null;
        }
        JSONObject body = new JSONObject();
        body.put("text", text);
        body.put("voice", sp.getString("voice", "Kore"));
        HttpURLConnection con = (HttpURLConnection) new URL(proxy.replaceAll("/+$", "") + "/tts").openConnection();
        con.setRequestMethod("POST");
        con.setConnectTimeout(10000);
        con.setReadTimeout(25000);
        con.setRequestProperty("content-type", "application/json");
        con.setRequestProperty("x-app-token", token);
        con.setDoOutput(true);
        OutputStream os = con.getOutputStream();
        os.write(body.toString().getBytes("UTF-8"));
        os.close();
        int rc = con.getResponseCode();
        if (rc != 200) {
            lastErr = "server " + rc;
            return null;
        }
        InputStream is = con.getInputStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[8192];
        int n;
        while ((n = is.read(buf)) > 0) bo.write(buf, 0, n);
        is.close();
        if (bo.size() < 1000) return null;
        File tmp = new File(dir, f.getName() + ".tmp");
        FileOutputStream fo = new FileOutputStream(tmp);
        fo.write(bo.toByteArray());
        fo.close();
        tmp.renameTo(f);
        return f;
    }

    static void play(File f) throws Exception {
        synchronized (lock) {
            if (player != null) {
                try { player.release(); } catch (Throwable e) { }
                player = null;
            }
            MediaPlayer mp = new MediaPlayer();
            mp.setAudioAttributes(new AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_ASSISTANT)
                .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH).build());
            mp.setDataSource(f.getAbsolutePath());
            mp.setOnCompletionListener(new MediaPlayer.OnCompletionListener() {
                public void onCompletion(MediaPlayer m) {
                    try { m.release(); } catch (Throwable e) { }
                }
            });
            mp.prepare();
            mp.start();
            player = mp;
        }
    }
