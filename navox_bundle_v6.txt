import android.Manifest;

import android.content.Intent;
import android.content.pm.PackageManager;

import android.net.Uri;

import android.view.View;
import android.webkit.JavascriptInterface;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import org.json.JSONObject;

public class MainActivity extends Activity {
    WebView web;

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        getWindow().setStatusBarColor(Color.parseColor("#FCFBF8"));
        getWindow().setNavigationBarColor(Color.parseColor("#FCFBF8"));
        web = new WebView(this);
        web.setBackgroundColor(Color.parseColor("#FCFBF8"));
        setContentView(web);
        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setAllowFileAccess(true);
        s.setAllowUniversalAccessFromFileURLs(true);
        web.addJavascriptInterface(new Bridge(), "NAVOX");
        web.setWebViewClient(new WebViewClient() {
            @Override
            public boolean shouldOverrideUrlLoading(WebView v, String url) {
                if (url.startsWith("file:")) return false;
                startActivity(new Intent(Intent.ACTION_VIEW, Uri.parse(url)));
                return true;
            }
        });
        web.loadUrl("file:///android_asset/index.html");
    }

    class Bridge {
        @JavascriptInterface
        public String exec(String text) {
            String r = CommandRunner.run(MainActivity.this, text);
            return r == null ? "" : r;
        }
        @JavascriptInterface
        public void speak(String text, String lang) {
            Voice.speak(MainActivity.this, text, lang);
        }
        @JavascriptInterface
        public void askMic() {
            if (Build.VERSION.SDK_INT >= 23) {
                requestPermissions(new String[]{Manifest.permission.RECORD_AUDIO}, 2);
            }
        }
    }
}

package com.navox.agent;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.pm.ServiceInfo;
import android.os.Build;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;
import android.os.PowerManager;
import android.widget.Toast;

public class ListenService extends Service {
    static boolean running = false;
    PowerManager.WakeLock wl;
    WakeEngine wake;

    @Override
    public IBinder onBind(Intent i) { return null; }

    @Override
    public int onStartCommand(Intent i, int flags, int id) {
        startFg();
        running = true;
        if (wl == null) {
            PowerManager pm = (PowerManager) getSystemService(Context.POWER_SERVICE);
            wl = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "navox:listen");
            wl.acquire();
        }
        if (wake == null) {
            wake = new WakeEngine(this, new WakeEngine.Callback() {
                public void onReady() { }
                public void onWake() {
                    Toast.makeText(ListenService.this, "NAVOX Triggered Instantly!", Toast.LENGTH_SHORT).show();
                }
                public void onFail() { }
            });
            wake.init();
        }
        return START_STICKY;
    }

    void startFg() {
        String ch = "navox_listen";
        NotificationManager nm = (NotificationManager) getSystemService(Context.NOTIFICATION_SERVICE);
        if (Build.VERSION.SDK_INT >= 26) {
            nm.createNotificationChannel(new NotificationChannel(ch, "NAVOX v6 active", NotificationManager.IMPORTANCE_LOW));
        }
        Notification.Builder b = Build.VERSION.SDK_INT >= 26 ? new Notification.Builder(this, ch) : new Notification.Builder(this);
        b.setContentTitle("NAVOX v6 Running");
        b.setContentText("Micro-second wake active");
        b.setSmallIcon(R.drawable.ic_notif);
        Notification n = b.build();
        if (Build.VERSION.SDK_INT >= 30) startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE);
        else startForeground(1, n);
    }

    @Override
    public void onDestroy() {
        running = false;
        if (wake != null) wake.destroy();
        if (wl != null && wl.isHeld()) wl.release();
        super.onDestroy();
    }
}

package com.navox.agent;

import android.content.Context;
import java.io.IOException;
import java.util.regex.Pattern;
import org.json.JSONObject;
import org.vosk.Model;
import org.vosk.Recognizer;
import org.vosk.android.RecognitionListener;
import org.vosk.android.SpeechService;
import org.vosk.android.StorageService;

public class WakeEngine implements RecognitionListener {
    public interface Callback {
        void onReady();
        void onWake();
        void onFail();
    }

    static final String GRAMMAR = "[\"hey nav\", \"hey nova\", \"hey nave\", \"[unk]\"]";
    static final Pattern WAKE = Pattern.compile("\\b(hey|hi|ok)\\s+(nav|nova|nave)\\b", Pattern.CASE_INSENSITIVE);

    final Context ctx;
    final Callback cb;
    Model model;
    Recognizer rec;
    SpeechService ss;
    boolean dead = false;

    public WakeEngine(Context c, Callback cb) {
        this.ctx = c;
        this.cb = cb;
    }

    public void init() {
        StorageService.unpack(ctx, "model", "model",
            new StorageService.Callback<Model>() {
                public void onComplete(Model m) {
                    if (dead) return;
                    model = m;
                    try {
                        rec = new Recognizer(model, 16000.0f, GRAMMAR);
                        resume();
                    } catch (Exception e) { cb.onFail(); }
                }
            },
            new StorageService.Callback<IOException>() {
                public void onComplete(IOException e) { cb.onFail(); }
            });
    }

    public void resume() {
        if (dead || rec == null) return;
        try {
            if (ss != null) { ss.stop(); ss.shutdown(); }
            rec.reset();
            ss = new SpeechService(rec, 16000.0f);
            ss.startListening(this);
            cb.onReady();
        } catch (Exception e) { }
    }

    public void destroy() {
        dead = true;
        if (ss != null) { try { ss.stop(); } catch(Exception e){} }
        try { if (rec != null) rec.close(); } catch(Exception e){}
        try { if (model != null) model.close(); } catch(Exception e){}
    }

    @Override
    public void onPartialResult(String json) {
        try {
            JSONObject o = new JSONObject(json);
            if (WAKE.matcher(o.optString("partial", "")).find()) {
                cb.onWake();
            }
        } catch (Exception e) { }
    }

    @Override public void onResult(String s) { }
    @Override public void onFinalResult(String s) { }
    @Override public void onError(Exception e) { }
    @Override public void onTimeout() { }
}

package com.navox.agent;

import android.content.Context;
import android.content.Intent;
import android.net.Uri;

public class CommandRunner {
    public static String run(Context c, String raw) {
        String t = raw.toLowerCase().trim();
        if (t.startsWith("open instagram")) {
            try {
                Intent i = c.getPackageManager().getLaunchIntentForPackage("com.instagram.android");
                if (i != null) { i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK); c.startActivity(i); return "Instagram opened successfully."; }
            } catch(Exception e) {}
            return "Could not open Instagram.";
        }
        if (t.startsWith("search")) {
            String q = t.replace("search", "").trim();
            Intent i = new Intent(Intent.ACTION_VIEW, Uri.parse("https://www.google.com/search?q=" + Uri.encode(q)));
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            c.startActivity(i);
            return "Searching for " + q;
        }
        return null;
    }
}

package com.navox.agent;

import android.content.Context;
import android.speech.tts.TextToSpeech;
import java.util.Locale;

public class Voice {
    static TextToSpeech tts;
    static boolean ready = false;

    public static void speak(Context ctx, final String text, String lang) {
        if (text == null || text.isEmpty()) return;
        final Locale loc = "te-IN".equals(lang) ? new Locale("te", "IN") : new Locale("en", "IN");
        if (tts == null) {
            tts = new TextToSpeech(ctx.getApplicationContext(), new TextToSpeech.OnInitListener() {
                public void onInit(int status) {
                    if (status == TextToSpeech.SUCCESS) {
                        ready = true;
                        tts.setLanguage(loc);
                        tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "navox_v6");
                    }
                }
            });
        } else if (ready) {
            tts.setLanguage(loc);
            tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "navox_v6");
        }
    }
}
