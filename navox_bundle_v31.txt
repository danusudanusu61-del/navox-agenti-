                q[j] = Math.min(Math.min(p[j] + 1, q[j - 1] + 1), p[j - 1] + cost);
            }
            p = q;
        }
        return p[n];
    }

    static App best(String name) {
        String t = norm(name);
        String al = ALIAS.get(t);
        if (al != null) t = al;
        if (t.length() < 2) return null;
        App best = null;
        double bs = 0;
        for (App a : APPS) {
            double sc = 0;
            if (a.n.equals(t)) sc = 100;
            else if (t.length() >= 3 && (a.n.contains(t) || (a.n.length() >= 4 && t.contains(a.n)))) sc = 85 - Math.abs(a.n.length() - t.length());
            else if (t.length() >= 5 && a.np.contains(t)) sc = 78;
            else if (t.length() >= 4) {
                double sim = 1.0 - (double) lev(a.n, t) / Math.max(a.n.length(), t.length());
                if (sim >= 0.72) sc = sim * 80;
            }
            if (sc > bs) {
                bs = sc;
                best = a;
            }
        }
        return bs >= 58 ? best : null;
    }

        ensure(c);

        if (a == null) {
            loadApps(c);
            a = best(name);
        }
        return a;
    }

            if (a != null && launch(c, a.pkg)) return "Opening " + a.label + ".";
        }
        return null;
    }

    static AccessibilityNodeInfo findByHint(AccessibilityNodeInfo root, String[] hints) {
        List<AccessibilityNodeInfo> q = new ArrayList<AccessibilityNodeInfo>();
        q.add(root);
        int i = 0;
        while (i < q.size() && i < 600) {
            AccessibilityNodeInfo n = q.get(i++);
            if (n == null) continue;
            CharSequence d = n.getContentDescription();
            CharSequence t = n.getText();
            String s = ((d == null ? "" : d.toString()) + " " + (t == null ? "" : t.toString())).toLowerCase();
            if (s.length() < 60 && n.isVisibleToUser()) {
                for (String hnt : hints) {
                    if (s.contains(hnt)) return n;
                }
            }
            for (int k = 0; k < n.getChildCount(); k++) {
                AccessibilityNodeInfo ch = n.getChild(k);
                if (ch != null) q.add(ch);
            }
        }
        return null;

            }
        } catch (Exception e) { }
    }

    public void onPartialResult(String s) {
        if (cmdMode) return;
        check(s, false);
    }

    @Override
    public void onResult(String s) {
        if (cmdMode) {
            cmdResult(s, false);
            return;
        }
        check(s, true);
    }

    @Override
    public void onFinalResult(String s) {
        if (cmdMode) cmdResult(s, true);
    }

    @Override
    public void onError(Exception e) {
        if (cmdMode) {
            cmdFinish(null);
            return;
        }
        if (!dead) h.postDelayed(resumeR, 1500);
    }

    @Override
    public void onTimeout() {
        if (cmdMode) cmdFinish(null);
    }

    static Screen dump() {
        Screen s = new Screen();
        NavoxAccess svc = NavoxAccess.inst;
        AccessibilityNodeInfo root = svc == null ? null : svc.getRootInActiveWindow();

        s.pkg = String.valueOf(root.getPackageName());
        List<AccessibilityNodeInfo> q = new ArrayList<AccessibilityNodeInfo>();
        q.add(root);

        while (i < q.size() && i < 800 && s.nodes.size() < 90) {
            AccessibilityNodeInfo n = q.get(i++);
            if (n == null) continue;
            for (int k = 0; k < n.getChildCount(); k++) {
                AccessibilityNodeInfo ch = n.getChild(k);
                if (ch != null) q.add(ch);
            }
            if (!n.isVisibleToUser()) continue;
            CharSequence tx = n.getText();
            CharSequence ds = n.getContentDescription();
            String label = (tx != null && tx.length() > 0) ? tx.toString() : (ds != null ? ds.toString() : "");
            boolean act = n.isClickable() || n.isEditable() || n.isScrollable() || n.isCheckable();
            if (label.length() == 0 && !act) continue;
            if (n.isPassword()) label = "(password field)";
            String kind = n.isEditable() ? "input" : n.isScrollable() ? "scroll" : n.isClickable() ? "button" : "text";
            if (kind.equals("text") && label.length() == 0) continue;
            s.text.append("[").append(s.nodes.size()).append("] ").append(kind).append(": ").append(trim(label, labelMax));
            if (n.isChecked()) s.text.append(" (on)");
            s.text.append("\n");
            s.nodes.add(n);
            s.labels.add(label);
        }

/* Reads ALL visible inbox rows at once and summarizes them with one cloud request. */
public class MailRead {
    static final String SYS = "You are NAVOX AI, built by Naveen Kumar. You get a list of the newest emails from the user's Gmail inbox "
        + "(each line has sender, subject, time and a preview). Reply in the same language as the user's request, in at most 3 short spoken sentences "
        + "(under 70 words), plain text. Say how many emails there are, mention the important senders, and clearly say if any email is about "
        + "a job, job application, interview or application status, and whether it looks genuine or like a scam. Never invent emails.";

    static void run(Context c, String goal, String proxy, String token) throws Exception {
        Intent i = c.getPackageManager().getLaunchIntentForPackage("com.google.android.gm");
        if (i == null) {
            AgentRunner.emit(c, "Gmail is not installed on this phone.", true);
            return;
        }
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        c.startActivity(i);
        Thread.sleep(2500);
        List<String> rows = readRows();
        for (int tries = 0; tries < 3 && rows.size() < 3 && !AgentRunner.cancel; tries++) {
            if (tries == 1) {
                NavoxAccess.global(AccessibilityService.GLOBAL_ACTION_BACK);
            }
            Thread.sleep(1200);
            rows = readRows();
        }
        if (rows.isEmpty()) {
            AgentRunner.emit(c, "I could not read your inbox. Please open Gmail and try again.", true);
            return;
        }
        if (!Cloud.tryUse(c)) {
            AgentRunner.emit(c, "Daily cloud limit reached. I opened Gmail, please read it yourself.", true);
            return;
        }
        StringBuilder sb = new StringBuilder("User request: " + goal + "\nEmails:\n");
        for (int k = 0; k < rows.size(); k++) sb.append(k + 1).append(". ").append(rows.get(k)).append("\n");
        Audit.log(c, AgentRunner.curTrace, "MAIL_ROWS", String.valueOf(rows.size()));
        String summary = ask(proxy, token, sb.toString()).trim();
        if (summary.length() == 0) summary = "I read " + rows.size() + " emails but could not summarize them.";
        AgentRunner.lastWasDone = true;
        AgentRunner.emit(c, summary, true);
    }

    static List<String> readRows() {
        int old = AgentRunner.labelMax;
        AgentRunner.labelMax = 300;
        AgentRunner.Screen sc;
        try {
            sc = AgentRunner.dump();
        } finally {
            AgentRunner.labelMax = old;
        }
        List<String> out = new ArrayList<String>();
        if (sc == null) return out;
        for (int k = 0; k < sc.labels.size(); k++) {
            String t = sc.labels.get(k) == null ? "" : sc.labels.get(k).trim();
            if (t.length() < 25) continue;
            String low = t.toLowerCase();
            if (low.startsWith("search") || low.contains("navigation drawer")) continue;
            if (!out.contains(t)) out.add(t);
            if (out.size() >= 14) break;
        }
        return out;
    }

    static String ask(String proxy, String token, String user) throws Exception {
        JSONObject body = new JSONObject();
        body.put("system", SYS);
        JSONArray msgs = new JSONArray();
        JSONObject m = new JSONObject();
        m.put("role", "user");
        m.put("content", user);
        msgs.put(m);
        body.put("messages", msgs);
        HttpURLConnection con = (HttpURLConnection) new URL(proxy).openConnection();
        con.setRequestMethod("POST");
        con.setConnectTimeout(15000);
        con.setReadTimeout(45000);
        con.setRequestProperty("content-type", "application/json");
        con.setRequestProperty("x-app-token", token);
        con.setDoOutput(true);
        OutputStream os = con.getOutputStream();
        os.write(body.toString().getBytes("UTF-8"));
        os.close();
        int code = con.getResponseCode();
        InputStream is = code >= 400 ? con.getErrorStream() : con.getInputStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[4096];
        int n;
        while (is != null && (n = is.read(buf)) > 0) bo.write(buf, 0, n);
        JSONObject r = new JSONObject(bo.toString("UTF-8"));
        if (code >= 400) {
            JSONObject er = r.optJSONObject("error");
            throw new Exception(er != null ? er.optString("message", "HTTP " + code) : "HTTP " + code);
        }
        return r.getJSONArray("content").getJSONObject(0).optString("text", "");
    }

    public static void start(final Context c, final int delayMs) {
        final String proxy = c.getSharedPreferences("navox", 0).getString("proxy", "");
        final String token = c.getSharedPreferences("navox", 0).getString("ptoken", "");
        if (proxy.length() == 0) {
            Voice.speak(c, "AI is not set up in the app.", "en-IN");
            return;
        }
        if (NavoxAccess.inst == null) {
            Voice.speak(c, "Please turn on NAVOX in Accessibility settings so I can read the screen.", "en-IN");
            return;
        }
        new Thread(new Runnable() {
            public void run() {
                try {
                    Thread.sleep(delayMs);
                    List<String> rows = read();
                    if (rows.isEmpty()) {
                        AgentRunner.emit(c, "I could not read anything on this screen.", true);
                        return;
                    }
                    if (!Cloud.tryUse(c)) {
                        AgentRunner.emit(c, "Daily cloud limit reached, so I cannot explain the screen now.", true);
                        return;
                    }
                    String lang = c.getSharedPreferences("navox", 0).getString("lang", "en-IN");

                        + (lang.startsWith("te") ? "Reply in simple Telugu." : "Reply in simple English.");
                    StringBuilder sb = new StringBuilder("Screen items:\n");
                    for (int i = 0; i < rows.size(); i++) sb.append("- ").append(rows.get(i)).append("\n");

                    String out = ask(proxy, token, sys, sb.toString()).trim();
                    AgentRunner.lastWasDone = true;
                    AgentRunner.emit(c, out.length() == 0 ? "I could not explain this screen." : out, true);

                    AgentRunner.emit(c, "I could not read this screen right now.", true);

        }).start();
    }
