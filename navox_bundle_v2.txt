    static int lev(String a, String b) {
        int m = a.length(), n = b.length();
        int[] p = new int[n + 1];
        for (int j = 0; j <= n; j++) p[j] = j;
        for (int i = 1; i <= m; i++) {
            int[] q = new int[n + 1];
            q[0] = i;
            for (int j = 1; j <= n; j++) {
                int cost = a.charAt(i - 1) == b.charAt(j - 1) ? 0 : 1;
                q[j] = Math.min(Math.min(p[j] + 1, q[j - 1] + 1), p[j - 1] + cost);
            }
            p = q;
        }
        return p[n];
    }

    static App best(String name) {
        String t = norm(name);
        String al = ALIAS.get(t);
        if (al != null) t = al;
        if (t.length() < 2) return null;
        App best = null;
        double bs = 0;
        for (App a : APPS) {
            double sc = 0;
            if (a.n.equals(t)) sc = 100;
            else if (t.length() >= 3 && (a.n.contains(t) || (a.n.length() >= 4 && t.contains(a.n)))) sc = 85 - Math.abs(a.n.length() - t.length());

    }

            else if (t.matches(".*(airplane|aeroplane|flight mode).*")) w = "airplane";
            if (w != null) return NavoxAccess.toggle(c, w, want);
        }

        if (words <= 5 && t.matches(".*\\b(volume|louder|quieter|mute|unmute)\\b.*")) {

                return "Volume maximum.";
            }

                am.setStreamVolume(AudioManager.STREAM_MUSIC, 0, AudioManager.FLAG_SHOW_UI);
                return "Muted.";
            }

                adjust(am, AudioManager.ADJUST_RAISE, 3);
                return "Volume up.";
            }
            if (t.matches(".*\\b(down|decrease|lower|quieter|less|low)\\b.*")) {
                adjust(am, AudioManager.ADJUST_LOWER, 3);
                return "Volume down.";
            }
            am.adjustStreamVolume(AudioManager.STREAM_MUSIC, AudioManager.ADJUST_SAME, AudioManager.FLAG_SHOW_UI);
            return "Volume.";
        }

        if (t.matches("^(go )?home( screen)?$")) return sys(c, AccessibilityService.GLOBAL_ACTION_HOME, "Home.");
        if (t.matches("^(go )?back$")) return sys(c, AccessibilityService.GLOBAL_ACTION_BACK, "Back.");
        if (t.matches("^(show )?(recent apps?|recents)$")) return sys(c, AccessibilityService.GLOBAL_ACTION_RECENTS, "Recent apps.");
        if (t.matches("^lock( the)?( screen| phone)?$")) return sys(c, AccessibilityService.GLOBAL_ACTION_LOCK_SCREEN, "Locking.");
        if (t.matches("^(open |show )?notifications?( panel| bar)?$")) return sys(c, AccessibilityService.GLOBAL_ACTION_NOTIFICATIONS, "Notifications.");
        if (t.matches("^(open |show )?quick settings$")) return sys(c, AccessibilityService.GLOBAL_ACTION_QUICK_SETTINGS, "Quick settings.");

        m = Pattern.compile("(?:^|\\s)(?:call|dial)\\s*(\\+?\\d[\\d\\s-]{2,})").matcher(t);
        if (m.find()) {
            String num = m.group(1).replaceAll("[\\s-]", "");
            return start(c, new Intent(Intent.ACTION_DIAL, Uri.parse("tel:" + Uri.encode(num)))) ? "Dialer opened for " + num + "." : null;
        }

        m = Pattern.compile("^(?:sms|text|message|msg)\\s+(\\+?\\d[\\d-]{5,})\\s*(.*)$").matcher(t);
        if (m.find()) {
            Intent i = new Intent(Intent.ACTION_SENDTO, Uri.parse("smsto:" + Uri.encode(m.group(1).replace("-", ""))));
            i.putExtra("sms_body", m.group(2));
            return start(c, i) ? "Message ready. Tap send." : null;
        }

        Matcher d = Pattern.compile("(\\d+)\\s*(sec\\w*|min\\w*|nimish\\w*|hour\\w*|hr\\w*|gant\\w*)").matcher(t);
        if ((t.contains("timer") || t.contains("remind")) && d.find()) {
            int n = Integer.parseInt(d.group(1));
            String u = d.group(2);
            int sec = n * (u.startsWith("s") ? 1 : (u.startsWith("h") || u.startsWith("g")) ? 3600 : 60);
            Intent i = new Intent(AlarmClock.ACTION_SET_TIMER);
            i.putExtra(AlarmClock.EXTRA_LENGTH, sec);
            i.putExtra(AlarmClock.EXTRA_MESSAGE, "NAVOX");
            i.putExtra(AlarmClock.EXTRA_SKIP_UI, true);
            return start(c, i) ? "Timer set for " + n + " " + u + "." : null;
        }

        if (t.matches(".*\\balarm\\b.*")) {
            Matcher a = Pattern.compile("(\\d{1,2})(?::(\\d{2}))?\\s*(am|pm)?").matcher(t);
            if (a.find()) {
                int h = Integer.parseInt(a.group(1));
                int mi = a.group(2) == null ? 0 : Integer.parseInt(a.group(2));
                String p = a.group(3);
                if ("pm".equals(p) && h < 12) h += 12;
                if ("am".equals(p) && h == 12) h = 0;
                if (h < 24 && mi < 60) {
                    Intent i = new Intent(AlarmClock.ACTION_SET_ALARM);
                    i.putExtra(AlarmClock.EXTRA_HOUR, h);
                    i.putExtra(AlarmClock.EXTRA_MINUTES, mi);
                    i.putExtra(AlarmClock.EXTRA_MESSAGE, "NAVOX");
                    i.putExtra(AlarmClock.EXTRA_SKIP_UI, true);
                    return start(c, i) ? "Alarm set for " + String.format("%02d:%02d", h, mi) + "." : null;
                }
            }
        }

        String yq = ytQuery(t);
        if (yq != null) {
            String url = "https://www.youtube.com/results?search_query=" + Uri.encode(yq);
            if (!view(c, url, "com.google.android.youtube")) view(c, url, null);
            return "Searching YouTube for " + yq + ".";
        }

        String name = null;
        m = Pattern.compile("^(?:open|launch|start|run)\\s+(?:the\\s+)?(.+?)(?:\\s+app)?$").matcher(t);
        if (m.find()) {
            name = m.group(1);
        } else {
            m = Pattern.compile("^(.+?)\\s+open(?:\\s+chey\\w*)?$").matcher(t);
            if (m.find()) name = m.group(1);
        }
        if (name != null && !name.matches(".*\\b(and|search|play|find)\\b.*")) {
            App a = find(c, name);
            if (a != null && launch(c, a.pkg)) return "Opening " + a.label + ".";
        }
        return null;
    }

package com.navox.agent;

import android.accessibilityservice.AccessibilityService;
import android.content.Context;
import android.content.Intent;
import android.os.Handler;
import android.os.Looper;
import android.provider.Settings;
import android.view.accessibility.AccessibilityEvent;
import android.view.accessibility.AccessibilityNodeInfo;
import java.util.ArrayList;
import java.util.List;

public class NavoxAccess extends AccessibilityService {
    static volatile NavoxAccess inst = null;
    final Handler h = new Handler(Looper.getMainLooper());

    @Override
    protected void onServiceConnected() {
        inst = this;
    }

    @Override
    public void onAccessibilityEvent(AccessibilityEvent e) { }

    @Override
    public void onInterrupt() { }

    @Override
    public boolean onUnbind(Intent i) {
        inst = null;
        return super.onUnbind(i);
    }

    static boolean global(int a) {
        NavoxAccess s = inst;
        return s != null && s.performGlobalAction(a);
    }

    static boolean open(Context c, String action) {
        try {
            Intent i = new Intent(action);
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            c.startActivity(i);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    static String toggle(Context c, String what, int want) {
        String action;
        String label;
        if (what.equals("wifi")) {
            action = "android.settings.panel.action.WIFI";
            label = "Wi-Fi";
        } else if (what.equals("bluetooth")) {
            action = Settings.ACTION_BLUETOOTH_SETTINGS;
            label = "Bluetooth";
        } else {
            action = Settings.ACTION_AIRPLANE_MODE_SETTINGS;
            label = "Airplane mode";
        }
        boolean opened = open(c, action);
        if (!opened && what.equals("wifi")) opened = open(c, Settings.ACTION_WIFI_SETTINGS);
        if (!opened) return "Could not open " + label + " settings.";
        if (want == 2) return "Opened " + label + ". Say on or off to switch it.";
        NavoxAccess s = inst;
        if (s == null) return "Opened " + label + ". Turn on NAVOX in Accessibility settings to switch it automatically.";
        s.clickSwitch(want, 0);
        return label + (want == 1 ? " on." : want == 0 ? " off." : " toggled.");
    }

    void clickSwitch(final int want, final int attempt) {
        h.postDelayed(new Runnable() {
            public void run() {
                boolean done = false;
                try {
                    AccessibilityNodeInfo root = getRootInActiveWindow();
                    if (root != null) {
                        AccessibilityNodeInfo sw = findSwitch(root);
                        if (sw != null) {
                            boolean isOn = sw.isChecked();
                            if (want == -1 || (want == 1) != isOn) clickNode(sw);
                            done = true;
                        }
                    }
                } catch (Exception e) { }
                if (done) {
                    h.postDelayed(new Runnable() {
                        public void run() {
                            performGlobalAction(GLOBAL_ACTION_BACK);
                        }
                    }, 900);
                } else if (attempt < 8) {
                    clickSwitch(want, attempt + 1);
                }
            }
        }, attempt == 0 ? 900 : 450);
    }

    static AccessibilityNodeInfo findSwitch(AccessibilityNodeInfo root) {
        List<AccessibilityNodeInfo> q = new ArrayList<AccessibilityNodeInfo>();
        q.add(root);
        int i = 0;
        while (i < q.size() && i < 500) {
            AccessibilityNodeInfo n = q.get(i++);
            if (n == null) continue;
            if (n.isCheckable() && n.isVisibleToUser()) return n;
            for (int k = 0; k < n.getChildCount(); k++) {
                AccessibilityNodeInfo ch = n.getChild(k);
                if (ch != null) q.add(ch);
            }
        }
        return null;
    }

    static void clickNode(AccessibilityNodeInfo n) {
        AccessibilityNodeInfo p = n;
        int d = 0;
        while (p != null && d < 6) {
            if (p.isClickable()) {
                p.performAction(AccessibilityNodeInfo.ACTION_CLICK);
                return;
            }
            p = p.getParent();
            d++;
        }
    }
}

package com.navox.agent;

import android.content.Context;
import android.os.Handler;
import android.os.Looper;
import java.io.IOException;
import java.util.regex.Pattern;
import org.json.JSONObject;
import org.vosk.Model;
import org.vosk.Recognizer;
import org.vosk.android.RecognitionListener;
import org.vosk.android.SpeechService;
import org.vosk.android.StorageService;

public class WakeEngine implements RecognitionListener {
    public interface Callback {
        void onReady();

        void onWake();

        void onFail();
    }

    static final String GRAMMAR = "[\"hey nav\", \"hey nova\", \"hey navy\", \"hey knave\", \"hey neve\", \"hi nav\", \"ok nav\", \"okay nav\", \"[unk]\"]";
    static final Pattern WAKE = Pattern.compile("\\b(hey|hi|hello|ok|okay)\\s+(nav|nova|navy|knave|neve)\\b");

    final Context ctx;
    final Callback cb;
    final Handler h = new Handler(Looper.getMainLooper());
    Model model;
    Recognizer rec;
    SpeechService ss;
    boolean dead = false;
    boolean fired = false;
    boolean ready = false;
    int tries = 0;

    final Runnable resumeR = new Runnable() {
        public void run() {
            resume();
        }
    };

    public WakeEngine(Context c, Callback cb) {
        this.ctx = c;
        this.cb = cb;
    }

    public void init() {
        StorageService.unpack(ctx, "model", "model",
                new StorageService.Callback<Model>() {
                    public void onComplete(Model m) {
                        if (dead) return;
                        model = m;
                        try {
                            rec = new Recognizer(model, 16000.0f, GRAMMAR);
                            resume();
                        } catch (Exception e) {
                            cb.onFail();
                        }
                    }
                },
                new StorageService.Callback<IOException>() {
                    public void onComplete(IOException e) {
                        cb.onFail();
                    }
                });
    }

    void stopService() {
        if (ss != null) {
            try {
                ss.stop();
            } catch (Exception e) { }
            try {
                ss.shutdown();
            } catch (Exception e) { }
            ss = null;
        }
    }

    public void resume() {
        if (dead || rec == null) return;
        fired = false;
        try {
            stopService();
            rec.reset();
            ss = new SpeechService(rec, 16000.0f);
            ss.startListening(this);
            tries = 0;
            if (!ready) {
                ready = true;
                cb.onReady();
            }
        } catch (Exception e) {
            tries++;
            if (tries < 6) h.postDelayed(resumeR, 1500);
            else cb.onFail();
        }
    }

    public void pause() {
        h.removeCallbacks(resumeR);
        stopService();
    }

    public void destroy() {
        dead = true;
        pause();
        try {
            if (rec != null) rec.close();
        } catch (Exception e) { }
        try {
            if (model != null) model.close();
        } catch (Exception e) { }
    }

    void check(String json) {
        if (fired || dead) return;
        try {
            JSONObject o = new JSONObject(json);
            String t = o.optString("partial", o.optString("text", ""));
            if (WAKE.matcher(t).find()) {
                fired = true;
                cb.onWake();
            }
        } catch (Exception e) { }
    }

    @Override
    public void onPartialResult(String s) {
        check(s);
    }

    @Override
    public void onResult(String s) {
        check(s);
    }

    @Override
    public void onFinalResult(String s) { }

    @Override
    public void onError(Exception e) {
        if (!dead) h.postDelayed(resumeR, 1500);
    }

    @Override
    public void onTimeout() { }
}
